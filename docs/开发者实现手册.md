### 3.2 交易日判定逻辑

`python
def get_data_trading_day():
    """
    判定数据所属交易日
    
    逻辑：
    1. 周末 → 回溯到上周五
    2. 交易日09:30前 → 使用上一交易日
    3. 交易日09:30-15:00 → 使用上一交易日
    4. 交易日15:00后 → 使用当日
    
    返回：(日期字符串, 提示信息)
    """
    now = datetime.now()
    hour, minute = now.hour, now.minute
    
    # 判断是否工作日（简化版，仅排除周末）
    is_weekday = now.weekday() < 5
    
    if not is_weekday:
        # 周末，回溯到上周五
        days_back = now.weekday() - 4  # 周一=0, 周五=4
        return (now - timedelta(days=days_back)).strftime('%Y-%m-%d'), \
               "今日为非交易日，已回溯至最近交易日"
    
    # 工作日的时间判断
    current_minutes = hour * 60 + minute
    
    if current_minutes < 570:  # 09:30前
        return get_last_trading_day(now - timedelta(days=1)), \
               "今日未开盘，以下为上一交易日收盘数据"
    elif current_minutes < 900:  # 09:30-15:00
        return get_last_trading_day(now - timedelta(days=1)), \
               "今日盘中，以下为上一交易日收盘数据"
    else:  # 15:00后
        return now.strftime('%Y-%m-%d'), ""

def get_last_trading_day(date):
    """获取最近的交易日（向前回溯，跳过周末）"""
    for i in range(30):
        check_date = date - timedelta(days=i)
        if check_date.weekday() < 5:  # 周一到周五
            return check_date.strftime('%Y-%m-%d')
    return (datetime.now() - timedelta(days=1)).strftime('%Y-%m-%d')
`

#### 3.3.2 连板数解析

`python
def parse_consecutive_boards(popularity_tag):
    """
    解析连板数
    
    输入示例：
    - "2天2板" → "2连板（2天2板）"
    - "首板涨停" → "首板（当日）"
    - None / "" → "无"
    
    注意：禁止推测，必须严格按格式转换
    """
    import re
    
    if not popularity_tag:
        return "无"
    
    # 匹配 "N天M板" 格式
    match = re.search(r'(\d+)天(\d+)板', popularity_tag)
    if match:
        days, boards = match.groups()
        return f"{boards}连板（{days}天{boards}板）"
    
    # 匹配 "首板涨停"
    if '首板' in popularity_tag:
        return "首板（当日）"
    
    return "无"
`

#### 3.3.4 异动解读处理

`python
def get_anomaly_analysis(item):
    """
    获取异动解读（四级优先级）
    
    优先级：
    1. analyse_title（分析标题）- 最优
    2. analyse（分析正文）- 截取前25字
    3. topic.title（热点话题）- 兜底
    4. "无" - 最终兜底
    
    注意：禁止从concept_tag推测！
    """
    # 优先级1：analyse_title
    if item.get('analyse_title'):
        return item['analyse_title'][:30]
    
    # 优先级2：analyse（截取25字）
    if item.get('analyse'):
        text = item['analyse'].replace('\n', ' ').strip()
        return text[:30]
    
    # 优先级3：topic.title
    topic = item.get('topic')
    if topic and topic.get('title'):
        return topic['title'][:30]
    
    # 优先级4：无
    return "无"
`

# 内存缓存
cache = {}
CACHE_TTL = 600  # 10分钟

def get_cache_key(url):
    """生成缓存键"""
    import hashlib
    return hashlib.md5(url.encode()).hexdigest()

def get_cached_data(key):
    """获取缓存数据，超时返回None"""
    if key in cache:
        data, timestamp = cache[key]
        if time.time() - timestamp < CACHE_TTL:
            return data
        else:
            del cache[key]  # 过期删除
    return None

def set_cached_data(key, data):
    """设置缓存"""
    cache[key] = (data, time.time())

def get_minute_data(code):
    """
    获取分时数据
    
    接口：https://web.ifzq.gtimg.cn/appstock/app/minute/query
    
    参数：code=sh600664 或 code=sz000002
    
    返回结构：
    {
        "data": {
            "sh600664": {
                "data": {
                    "data": [
                        "0930 9.42 191254 180161268.50",  # 时间 价格 成交量 成交金额
                        "0931 9.41 150000 141000000.00",
                        ...
                    ]
                }
            }
        }
    }
    
    解析规则：
    - 按空格分割
    - 第1项：时间（HHMM格式，转为HH:MM）
    - 第2项：价格
    - 第3项：成交量
    - 第4项：成交金额（单位：元）
    """
    clean_code = code.replace('SH', '').replace('SZ', '')
    market_code = f"sh{clean_code}" if clean_code.startswith('6') else f"sz{clean_code}"
    
    url = f"https://web.ifzq.gtimg.cn/appstock/app/minute/query?code={market_code}"
    response = requests.get(url, timeout=15, verify=False)
    data = response.json()
    
    stock_data = data.get('data', {}).get(market_code, {}).get('data', {}).get('data', [])
    
    result = []
    for item in stock_data:
        parts = item.split(' ')
        if len(parts) >= 4:
            time_str = parts[0]
            if len(time_str) == 4:
                time_str = time_str[:2] + ':' + time_str[2:]
            result.append({
                'time': time_str,
                'price': float(parts[1]),
                'volume': int(parts[2]),
                'amount': float(parts[3])
            })
    
    return result
`

def get_daily_data(code):
    """
    获取日K线数据
    
    接口：https://quotes.sina.cn/cn/api/jsonp_v2.php
    
    参数：
    - symbol: sh600664 或 sz000002
    - scale: 240（日线）
    - datalen: 60（最近60天）
    
    返回格式（JSONP）：
    var xxx=[{"day":"2026-09-22","open":"9.02","close":"9.08","high":"9.08","low":"8.68","volume":"308205511"},...]
    
    解析规则：
    - 去掉 JSONP 包装 var xxx=(...);
    - 解析JSON数组
    - 每项包含：day, open, close, high, low, volume
    """
    clean_code = code.replace('SH', '').replace('SZ', '')
    market_code = f"sh{clean_code}" if clean_code.startswith('6') else f"sz{clean_code}"
    
    url = f"https://quotes.sina.cn/cn/api/jsonp_v2.php/var/CN_MarketDataService.getKLineData?symbol={market_code}&scale=240&ma=no&datalen=60"
    response = requests.get(url, timeout=15, verify=False)
    text = response.text
    
    # 解析JSONP
    start = text.find('(')
    end = text.rfind(')')
    klines = json.loads(text[start+1:end])
    
    result = []
    for item in klines:
        result.append({
            'date': item['day'],
            'open': float(item['open']),
            'close': float(item['close']),
            'high': float(item['high']),
            'low': float(item['low']),
            'volume': int(item['volume'])
        })
    
    return result
`

## 七、常见问题与解决方案

| 问题 | 原因 | 解决方案 |
|------|------|----------|
| 同花顺接口403 | 缺少Referer头 | 添加 Referer: https://www.10jqka.com.cn/ |
| 降级接口失败 | 需要POST请求 | 使用 requests.post() 而非 get() |
| 涨跌幅显示0% | 使用了错误的基准价 | 改用前一日收盘价计算 |
| K线图不显示 | JSONP格式解析错误 | 先去掉 var xxx=(...); 再解析 |
| 连板数显示错误 | 正则匹配不正确 | 使用 r'(\d+)天(\d+)板' 匹配 |
